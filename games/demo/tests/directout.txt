# Direct output (RenderConfig { msaa: 1, direct_output: true }): the main view goes straight into the
# canvas with the LDR pipeline set while the composite would only tonemap. Open the page with
# ?msaa=1&direct=1 first:
#   node tools/bridge.mjs launch --url "http://localhost:8080/?msaa=1&direct=1"
#   node tools/bridge.mjs run games/demo/tests/directout.txt
# Pixel check (direct vs composite, same settings) was done by hand: title screen mean diff 0.8/255;
# semi-transparent surfaces blend after the tonemap (a 50 % red cube over the sky reads redder).
reload
wait 1000
call render {"bloom": false, "fxaa": false, "depth_darkening": 0, "lens_flare": 0}
frames 3
expect stats.render.direct_output == true
expect stats.render.direct_blocked == ""
call render {"bloom": true}
frames 2
expect stats.render.direct_output == false
expect stats.render.direct_blocked == "bloom"
call render {"bloom": false, "direct_output": false}
frames 2
expect stats.render.direct_blocked == "off"
call render {"direct_output": true, "grain": 0.3}
frames 2
expect stats.render.direct_blocked == "grain"
call render {"grain": 0}
call spawn {"shape": "cube", "pos": [0, 2, 2], "size": 1.5, "color": "#ff4040", "body": "none", "name": "glass"}
call set {"name": "glass", "opacity": 0.5}
call particles {"preset": "fire", "pos": [1.5, 1, 3]}
frames 10
expect stats.render.direct_output == true
screen play
wait 1000
expect stats.render.direct_blocked == "water"
