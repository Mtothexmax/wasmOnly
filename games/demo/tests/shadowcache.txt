# Shadow cache: a still scene renders no shadow pass, moving renders cascade 0 (and refreshes the others).
# (something animated near the camera keeps cascade 0 busy in the demo, so "still" means >= 2 of 4 cached)
# Run: node tools/bridge.mjs run games/demo/tests/shadowcache.txt
reload
screen play
# glTF models (the skinned critter) finish loading asynchronously; their first pose re-renders the
# shadows once, so wait for them before calling the scene still
wait_for assets.models_pending.length == 0 10000
pause
call render {"shadow_cache":true,"particles":false}
call sky {"time":15.5,"day_length":0}
# the critter skin keeps settling for up to ~25 frames after its load (also before the surface hook)
frames 60
# nothing changes while paused (a morph-weight rounding flip used to re-deform the critter every frame)
expect gpu_scene.changed.length == 0
expect stats.render.shadow_passes_cached >= 2
expect stats.render.shadow_passes_rendered <= 2
down KeyW
step 20
frames 2
up KeyW
call render {"shadow_cache":false}
frames 3
expect stats.render.shadow_passes_cached == 0
expect stats.render.shadow_passes_rendered >= 1
call render {"shadow_cache":true}

# a moving caster re-renders the cascade its shadow lands in every frame (no delayed refresh)
# log splits put the ball 7 m away into cascade 2. Real frames, not "step": step renders its own frame.
call render {"shadow_distance":200,"shadow_split_lambda":1}
resume
down KeyW
frames 20
expect stats.render.shadow_cascades_rendered.2 == true
frames 1
expect stats.render.shadow_cascades_rendered.2 == true
frames 1
expect stats.render.shadow_cascades_rendered.2 == true
up KeyW
call render {"shadow_distance":160,"shadow_split_lambda":0.8,"particles":true}
